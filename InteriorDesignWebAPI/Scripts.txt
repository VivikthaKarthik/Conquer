CREATE           VIEW [dbo].[vwProjects]
AS
Select 
	P.Id, 
	P.[Name],
	C.Id AS ClientId,
	C.[Name] AS ClientName,
	S.Id AS StatusId,
	S.[Name] AS [Status],
	HT.ID AS HouseTypeId,
	HT.[Name] AS HouseType,
	HT.ImagePath,
	MT.ID AS MaterialTypeId,
	MT.[Name] AS MaterialType,
	OT.ID AS OuterFrameTypeId,
	OT.[Name] AS OuterFrameType,
	P.StartDate,
	P.EndDate
from Project P
Inner Join Client C ON C.Id = P.ClientId
Inner Join [Status] S ON S.Id = P.StatusId
Inner Join HouseType HT ON HT.Id = P.HouseTypeId
Inner Join MaterialType MT ON MT.Id = P.MaterialTypeId
Inner Join OuterFrameType OT ON OT.Id = P.OuterFrameTypeId

GO



CREATE     VIEW [dbo].[vwRooms]
AS

Select 
	R.ID, 
	RT.Id AS RoomTypeId,
	RT.[Name],
	P.Id AS ProjectId, 
	P.[Name] AS ProjectName,
	MT.ID AS MaterialTypeId,
	MT.[Name] AS MaterialType,
	OT.ID AS OuterFrameTypeId,
	OT.[Name] AS OuterFrameType
from Room R
INNER JOIN RoomType RT ON RT.ID = R.RoomTypeId
INNER JOIN Project P ON P.ID = R.ProjectId
Inner Join MaterialType MT ON MT.Id = R.MaterialTypeId
Inner Join OuterFrameType OT ON OT.Id = R.OuterFrameTypeId

GO


CREATE             VIEW [dbo].[vwWorkOrders]
AS
Select 
	W.Id, 
	W.[Name],
	WI.ID AS WorkOrderItemId,
	ISNULL(WI.[Name], '') AS WorkOrderItem,
	W.WorkOrderType,
	R.Id AS RoomId,
	RT.Id AS RoomTypeId,
	RT.[Name] AS Room,
	W.DesignType,
	W.OuterFrameType,
	W.MaterialType,
	W.Height,
	W.Width,
	W.SuppressCalculation,
	W.Amount
from [WorkOrder] W
LEFT Join WorkOrderItem WI ON WI.Id = W.WorkOrderItemId
Inner Join Room R ON R.Id = W.RoomId
INNER JOIN RoomType RT ON RT.ID = R.RoomTypeId

GO
